## 1. Producer-Consumer Problem Using Binary Semaphore

### Question
Solve the producer-consumer problem for a single piece of data using a binary semaphore.

### Code

```c
#include <stdio.h>
#include <pthread.h>
#include <semaphore.h>

int data;
sem_t empty, full;

void *producer(void *arg) {
    data = 10;
    printf("Produced: %d\n", data);
    sem_post(&full);
    return NULL;
}

void *consumer(void *arg) {
    sem_wait(&full);
    printf("Consumed: %d\n", data);
    sem_post(&empty);
    return NULL;
}

int main() {
    pthread_t p, c;
    sem_init(&empty, 0, 1);
    sem_init(&full, 0, 0);

    pthread_create(&p, NULL, producer, NULL);
    pthread_create(&c, NULL, consumer, NULL);

    pthread_join(p, NULL);
    pthread_join(c, NULL);
    return 0;
}
````

### How to Run

```bash
gcc q1.c -o q1 -pthread
./q1
```

### Output

```text
Produced: 10
Consumed: 10
```

## 2. Producer-Consumer Problem with Two Consumers

### Question

Solve the producer-consumer problem for a single piece of data with two consumer processes/threads using semaphores.

### Code

```c
#include <stdio.h>
#include <pthread.h>
#include <semaphore.h>

int data, consumed = 0;
sem_t full, mutex;

void *producer(void *arg) {
    data = 10;
    printf("Produced: %d\n", data);
    sem_post(&full);
    sem_post(&full);
    return NULL;
}

void *consumer(void *arg) {
    sem_wait(&full);
    sem_wait(&mutex);
    printf("Consumer %ld consumed: %d\n", (long)arg, data);
    sem_post(&mutex);
    return NULL;
}

int main() {
    pthread_t p, c1, c2;
    sem_init(&full, 0, 0);
    sem_init(&mutex, 0, 1);

    pthread_create(&p, NULL, producer, NULL);
    pthread_create(&c1, NULL, consumer, (void *)1);
    pthread_create(&c2, NULL, consumer, (void *)2);

    pthread_join(p, NULL);
    pthread_join(c1, NULL);
    pthread_join(c2, NULL);
    return 0;
}
```

### How to Run

```bash
gcc q2.c -o q2 -pthread
./q2
```

### Output

```text
Produced: 10
Consumer 1 consumed: 10
Consumer 2 consumed: 10
```

> Note: Both consumers receive the same single piece of data. The semaphore is posted twice so that both consumer threads can demonstrate consumption.

## 3. Producer-Consumer Using a Bounded Array

### Question

Solve the producer problem using a bounded-length array.

### Code

```c
#include <stdio.h>
#include <pthread.h>
#include <semaphore.h>

#define N 5

int buf[N], in = 0, out = 0;
sem_t empty, full, mutex;

void *producer(void *arg) {
    for (int i = 1; i <= 10; i++) {
        sem_wait(&empty);
        sem_wait(&mutex);

        buf[in] = i;
        printf("Produced: %d\n", i);
        in = (in + 1) % N;

        sem_post(&mutex);
        sem_post(&full);
    }
    return NULL;
}

void *consumer(void *arg) {
    for (int i = 1; i <= 10; i++) {
        sem_wait(&full);
        sem_wait(&mutex);

        printf("Consumed: %d\n", buf[out]);
        out = (out + 1) % N;

        sem_post(&mutex);
        sem_post(&empty);
    }
    return NULL;
}

int main() {
    pthread_t p, c;

    sem_init(&empty, 0, N);
    sem_init(&full, 0, 0);
    sem_init(&mutex, 0, 1);

    pthread_create(&p, NULL, producer, NULL);
    pthread_create(&c, NULL, consumer, NULL);

    pthread_join(p, NULL);
    pthread_join(c, NULL);
    return 0;
}
```

### How to Run

```bash
gcc q3.c -o q3 -pthread
./q3
```

### Output

```text
Produced: 1
Produced: 2
Consumed: 1
Produced: 3
Consumed: 2
...
Produced: 10
Consumed: 10
```

> The exact order can vary because producer and consumer execute concurrently.

## 4. Pthread Condition Variables

### Question

Write a program using Pthread condition-variable routines. The main routine creates three threads. Two threads update a `count` variable, while the third waits until `count` reaches 500.

### Code

```c
#include <stdio.h>
#include <pthread.h>

int count = 0;
pthread_mutex_t m = PTHREAD_MUTEX_INITIALIZER;
pthread_cond_t c = PTHREAD_COND_INITIALIZER;

void *work(void *arg) {
    for (int i = 0; i < 250; i++) {
        pthread_mutex_lock(&m);
        count++;
        if (count >= 500)
            pthread_cond_signal(&c);
        pthread_mutex_unlock(&m);
    }
    return NULL;
}

void *waiter(void *arg) {
    pthread_mutex_lock(&m);
    while (count < 500)
        pthread_cond_wait(&c, &m);
    printf("Count reached: %d\n", count);
    pthread_mutex_unlock(&m);
    return NULL;
}

int main() {
    pthread_t t1, t2, t3;

    pthread_create(&t1, NULL, work, NULL);
    pthread_create(&t2, NULL, work, NULL);
    pthread_create(&t3, NULL, waiter, NULL);

    pthread_join(t1, NULL);
    pthread_join(t2, NULL);
    pthread_join(t3, NULL);
    return 0;
}
```

### How to Run

```bash
gcc q4.c -o q4 -pthread
./q4
```

### Output

```text
Count reached: 500
```
